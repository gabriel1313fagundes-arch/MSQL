import sqlite3

conexao = None

try:
    # Abre o banco da biblioteca e cria o cursor para executar SQL
    conexao = sqlite3.connect("biblioteca.db")
    cursor = conexao.cursor()

    # Cria a tabela livros
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS livros (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            titulo TEXT NOT NULL,
            autor TEXT,
            ano INTEGER,
            -- 1 = disponível e 0 = emprestado
            disponivel INTEGER NOT NULL CHECK (disponivel IN (0, 1))
        )
    """)

    # 1. CREATE: cadastra cinco livros se a tabela estiver vazia
    cursor.execute("SELECT COUNT(*) FROM livros")
    quantidade = cursor.fetchone()[0]

    if quantidade == 0:
        livros = [
            ("Dom Casmurro", "Machado de Assis", 1899, 1),
            ("O Cortiço", "Aluísio Azevedo", 1890, 1),
            ("Jogos Vorazes", "Suzanne Collins", 2008, 1),
            ("A Culpa é das Estrelas", "John Green", 2012, 0),
            ("Extraordinário", "R. J. Palacio", 2012, 1)
        ]

        cursor.executemany(
            "INSERT INTO livros (titulo, autor, ano, disponivel) VALUES (?, ?, ?, ?)",
            livros
        )
        conexao.commit()

    # 2. READ: mostra todos os livros e a situação de cada um
    print("\nTodos os livros:")
    cursor.execute("SELECT id, titulo, autor, ano, disponivel FROM livros ORDER BY id")

    for id_livro, titulo, autor, ano, disponivel in cursor.fetchall():
        if disponivel == 1:
            situacao = "1 = disponível"
        else:
            situacao = "0 = emprestado"

        print(f"{id_livro} - {titulo} | {autor} | {ano} | {situacao}")

    # Mostra apenas os livros disponíveis
    print("\nLivros disponíveis:")
    cursor.execute("SELECT titulo FROM livros WHERE disponivel = ?", (1,))

    for livro in cursor.fetchall():
        print(livro[0])

    # 3. UPDATE: simula o empréstimo de Jogos Vorazes
    cursor.execute(
        "UPDATE livros SET disponivel = ? WHERE titulo = ? AND disponivel = ?",
        (0, "Jogos Vorazes", 1)
    )
    conexao.commit()

    if cursor.rowcount > 0:
        print("\nEmpréstimo registrado: Jogos Vorazes.")

    # 4. DELETE: remove o livro doado para outra escola
    cursor.execute("DELETE FROM livros WHERE titulo = ?", ("O Cortiço",))
    conexao.commit()

    if cursor.rowcount > 0:
        print("\nO Cortiço foi doado e removido da biblioteca.")

    # 5. EXTRA: lista os livros publicados depois de 2000
    print("\nLivros publicados depois de 2000:")
    cursor.execute(
        "SELECT titulo, ano FROM livros WHERE ano > ? ORDER BY ano, titulo",
        (2000,)
    )

    for titulo, ano in cursor.fetchall():
        print(f"{titulo} - {ano}")

    # Menu: permite escolher um livro e consultar a situação atual no banco
    while True:
        print("\nEscolha um livro pelo ID:")
        cursor.execute("SELECT id, titulo FROM livros ORDER BY id")

        for id_livro, titulo in cursor.fetchall():
            print(f"{id_livro} - {titulo}")

        print("0 - Sair")
        escolha = input("Digite o ID do livro: ").strip()

        if escolha == "0":
            print("Consulta encerrada.")
            break

        # Busca apenas o livro que a pessoa escolheu
        cursor.execute(
            "SELECT titulo, disponivel FROM livros WHERE id = ?",
            (escolha,)
        )
        livro = cursor.fetchone()

        if livro is None:
            print("Livro não encontrado. Escolha um ID da lista.")
        else:
            titulo, disponivel = livro

            if disponivel == 1:
                print(f"{titulo} está disponível (1).")
            else:
                print(f"{titulo} está emprestado (0).")

except sqlite3.Error as erro:
    print("Erro no banco de dados:", erro)

finally:
    # Fecha a conexão ao sair do programa
    if conexao is not None:
        conexao.close()








import sqlite3

# Abre o banco de dados. Se o arquivo ainda não existir, o SQLite cria.
banco = sqlite3.connect("petshop.db")
cursor = banco.cursor()

# Cria a tabela. O id é automático e o nome é obrigatório.
cursor.execute("""
    CREATE TABLE IF NOT EXISTS pets (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        nome TEXT NOT NULL,
        especie TEXT,
        tutor TEXT,
        idade INTEGER
    )
""")

# 1. CREATE - Cadastra os cinco pets na primeira execução.
cursor.execute("SELECT COUNT(*) FROM pets")
quantidade = cursor.fetchone()[0]

if quantidade == 0:
    pets = [
        ("Rex", "cachorro", "João", 4),
        ("Luna", "gato", "Ana", 2),
        ("Pipoca", "coelho", "Pedro", 1),
        ("Mel", "cachorro", "Bia", 3),
        ("Louro", "papagaio", "Lucas", 6)
    ]

    cursor.executemany("""
        INSERT INTO pets (nome, especie, tutor, idade)
        VALUES (?, ?, ?, ?)
    """, pets)
    banco.commit()  # Salva os cadastros.
    print("5 pets cadastrados!")

# 2. READ - Lista todos os pets e depois apenas os cachorros.
print("\nTodos os pets:")
cursor.execute("SELECT id, nome, especie, tutor, idade FROM pets ORDER BY id")

for id_pet, nome, especie, tutor, idade in cursor.fetchall():
    print(f"{id_pet} - {nome} | {especie} | Tutor: {tutor} | Idade: {idade} ano(s)")

print("\nApenas os cachorros:")
cursor.execute("SELECT nome FROM pets WHERE especie = ? ORDER BY id", ("cachorro",))

for linha in cursor.fetchall():
    print(linha[0])

# 3. UPDATE - Troca a tutora da Luna de Ana para Carla.
cursor.execute("""
    UPDATE pets SET tutor = ?
    WHERE nome = ? AND especie = ? AND tutor = ?
""", ("Carla", "Luna", "gato", "Ana"))

if cursor.rowcount > 0:
    print("\nCarla agora é a tutora da Luna.")

# 4. DELETE - Exclui o coelho Pipoca, que não é mais atendido.
cursor.execute("""
    DELETE FROM pets
    WHERE nome = ? AND especie = ? AND tutor = ?
""", ("Pipoca", "coelho", "Pedro"))

if cursor.rowcount > 0:
    print("Pipoca foi excluído do cadastro.")

banco.commit()  # Salva a troca de tutor e a exclusão.

# 5. EXTRA - Mostra somente o nome e o tutor dos pets restantes.
print("\nNome do pet | Tutor")
cursor.execute("SELECT nome, tutor FROM pets ORDER BY id")

for nome, tutor in cursor.fetchall():
    print(f"{nome} | {tutor}")


except sqlite3.Error as erro:
    print("Erro no banco de dados:", erro)

finally:
